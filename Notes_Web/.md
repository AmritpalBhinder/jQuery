import axios from "axios";

const API = axios.create({
  baseURL: "http://localhost:3001/tasks"
});

export const getTasks = () => API.get("/");
export const createTask = (task) => API.post("/", task);
export const deleteTask = (id) => API.delete(`/${id}`);
export const updateTask = (id,data) => API.put(`/${id}`,data);

import React, { useEffect, useState } from "react";
import { getTasks, createTask, deleteTask, updateTask } from "./services/api";

import Navbar from "./components/Navbar";
import TaskForm from "./components/TaskForm";
import TaskTable from "./components/TaskTable";
import Stats from "./components/Stats";
import SearchBar from "./components/SearchBar";

function App() {

const [tasks,setTasks] = useState([]);
const [search,setSearch] = useState("");

const fetchTasks = async()=>{
const res = await getTasks();
setTasks(res.data);
};

useEffect(()=>{
fetchTasks();
},[]);

const addTask = async(task)=>{
await createTask(task);
fetchTasks();
};

const removeTask = async(id)=>{
await deleteTask(id);
fetchTasks();
};

const toggleStatus = async(task)=>{

await updateTask(task.id,{
...task,
status: task.status === 1 ? 0 : 1
});

fetchTasks();

};

const filteredTasks = tasks.filter(t =>
t.task.toLowerCase().includes(search.toLowerCase()) ||
t.username.toLowerCase().includes(search.toLowerCase())
);

return (

<div>

<Navbar/>

<div className="container">

<h1>My Todo List</h1>

<Stats tasks={tasks}/>

<SearchBar setSearch={setSearch}/>

<TaskForm addTask={addTask}/>

<TaskTable
tasks={filteredTasks}
removeTask={removeTask}
toggleStatus={toggleStatus}
/>

</div>

</div>

);

}

export default App;



function Navbar(){

return(
<div className="navbar">
TASK MANAGER
</div>
)

}

export default Navbar;


function Stats({tasks}){

const total = tasks.length;
const done = tasks.filter(t=>t.status===1).length;
const pending = tasks.filter(t=>t.status===0).length;

return(

<div className="stats">

<div className="card">
<h2>{total}</h2>
<p>Total</p>
</div>

<div className="card">
<h2>{pending}</h2>
<p>Pending</p>
</div>

<div className="card">
<h2>{done}</h2>
<p>Done</p>
</div>

</div>

)

}

export default Stats;


function SearchBar({setSearch}){

return(

<input
className="search"
placeholder="Search tasks or username..."
onChange={(e)=>setSearch(e.target.value)}
/>

)

}

export default SearchBar;


import {useState} from "react";

function TaskForm({addTask}){

const [task,setTask] = useState("");
const [username,setUsername] = useState("");
const [date,setDate] = useState("");
const [type,setType] = useState("office");

const submit=(e)=>{

e.preventDefault();

if(task.length<3) return alert("Task min 3 characters");
if(username==="") return alert("Username required");

addTask({
task,
username,
date,
taskType:type,
status:0
});

setTask("");
setUsername("");
setDate("");

}

return(

<form onSubmit={submit} className="form">

<textarea
placeholder="Task"
value={task}
onChange={(e)=>setTask(e.target.value)}
/>

<input
placeholder="Username"
value={username}
onChange={(e)=>setUsername(e.target.value)}
/>

<input
type="date"
value={date}
onChange={(e)=>setDate(e.target.value)}
/>

<select onChange={(e)=>setType(e.target.value)}>

<option value="office">Office</option>
<option value="personal">Personal</option>
<option value="family">Family</option>
<option value="friends">Friends</option>
<option value="other">Other</option>

</select>

<button>Add Task</button>

</form>

)

}

export default TaskForm;


import TaskRow from "./TaskRow";

function TaskTable({tasks,removeTask,toggleStatus}){

return(

<table>

<thead>

<tr>
<th>#</th>
<th>Task</th>
<th>User</th>
<th>Date</th>
<th>Type</th>
<th>Status</th>
<th>Action</th>
</tr>

</thead>

<tbody>

{tasks.map((task,i)=>(

<TaskRow
key={task.id}
task={task}
index={i}
removeTask={removeTask}
toggleStatus={toggleStatus}
/>

))}

</tbody>

</table>

)

}

export default TaskTable;


function TaskRow({task,index,removeTask,toggleStatus}){

const colors={
office:"red",
personal:"orange",
family:"green",
friends:"cyan",
other:"gray"
}

return(

<tr>

<td>{index+1}</td>

<td>{task.task}</td>

<td>@{task.username}</td>

<td>{task.date}</td>

<td style={{color:colors[task.taskType]}}>
{task.taskType}
</td>

<td>

<button onClick={()=>toggleStatus(task)}>
{task.status===1?"Done":"Pending"}
</button>

</td>

<td>

<button onClick={()=>removeTask(task.id)}>
Delete
</button>

</td>

</tr>

)

}

export default TaskRow;
